      if (!Number.isFinite(value)) return '0:00';

      const minutes = Math.floor(value / 60);

      const seconds = Math.floor(value % 60).toString().padStart(2, '0');

      return minutes + ':' + seconds;

    }

    function dateText(value) {

      const date = new Date(value);

      return Number.isNaN(date.getTime()) ? '' : date.toLocaleDateString(undefined, {month:'short',day:'numeric',year:'numeric'});

    }

    function showView(view) {

      const upload = view === 'upload';

      galleryView.style.display = upload ? 'none' : 'block';

      uploadView.style.display = upload ? 'block' : 'none';

      tabs.forEach(tab => tab.setAttribute('aria-selected', String(tab.dataset.view === view)));

      if (upload) window.scrollTo({top: root.getBoundingClientRect().top + window.scrollY, behavior:'smooth'});

    }

    

    tabs.forEach(tab => tab.addEventListener('click', () => showView(tab.dataset.view)));

    search.addEventListener('input', renderFilter);


    function setPlayOverlay(button, visible) { button.classList.toggle('rh-show', visible); }

    function pauseAll(except) {

      root.querySelectorAll('.rh-video').forEach(video => { if (video !== except) video.pause(); });

    }

    function attemptPlay(video, overlay) {

      pauseAll(video);

      video.muted = true;

      activeVideo = video;

      const promise = video.play();

      if (promise && promise.catch) promise.catch(() => setPlayOverlay(overlay, true));

    }

    function createIcon(name) {

      const paths = {

        play:'<path d="m8 5 12 7-12 7V5Z" fill="currentColor" stroke="none"/>',

        pause:'<path d="M8 5h3v14H8zM15 5h3v14h-3z" fill="currentColor" stroke="none"/>',

        volume:'<path d="M4 10v4h3l4 3V7l-4 3H4Z"/><path d="M15 9a5 5 0 0 1 0 6M17.5 6.5a9 9 0 0 1 0 11"/>',

        mute:'<path d="M4 10v4h3l4 3V7l-4 3H4Z"/><path d="m16 9 5 6m0-6-5 6"/>',

        full:'<path d="M8 3H3v5m0-5 6 6m7-6h5v5m0-5-6 6M3 16v5h5m-5 0 6-6m10 1v5h-5m5 0-6-6"/>'

      };

      const svg = document.createElementNS('http://www.w3.org/2000/svg','svg');

      svg.setAttribute('viewBox','0 0 24 24'); svg.setAttribute('fill','none'); svg.setAttribute('stroke','currentColor'); svg.setAttribute('stroke-width','2'); svg.setAttribute('stroke-linecap','round'); svg.setAttribute('stroke-linejoin','round'); svg.innerHTML = paths[name];

      return svg;

    }

    function makeButton(className, label, icon) {

      const button = document.createElement('button'); button.type='button'; button.className=className; button.setAttribute('aria-label',label); button.title=label; if(icon) button.append(createIcon(icon)); return button;

    }

    function createCard(item) {

      const article = document.createElement('article'); article.className='rh-card'; article.dataset.id=String(item.id);

      const frame = document.createElement('div'); frame.className='rh-frame';

      const video = document.createElement('video'); video.className='rh-video'; video.src=item.url; video.playsInline=true; video.muted=true; video.loop=true; video.preload='metadata'; video.setAttribute('aria-label',item.title || 'Uploaded video');

      const playOverlay = document.createElement('button'); playOverlay.type='button'; playOverlay.className='rh-play-overlay'; playOverlay.setAttribute('aria-label','Play video');

      const playCircle=document.createElement('span');playCircle.className='rh-play-circle';playCircle.append(createIcon('play'));playOverlay.append(playCircle);

      const spinner=document.createElement('span');spinner.className='rh-spinner';spinner.hidden=true;spinner.setAttribute('aria-label','Loading video');

      const fullscreen=makeButton('rh-full','Open fullscreen','full');

      const tools=document.createElement('div');tools.className='rh-tools';

      const playButton=makeButton('rh-control','Play','play');

      const timeline=document.createElement('input');timeline.type='range';timeline.min='0';timeline.max='1000';timeline.value='0';timeline.className='rh-progress';timeline.setAttribute('aria-label','Seek video');

      const time=document.createElement('span');time.className='rh-time';time.textContent='0:00';

      const muteButton=makeButton('rh-control','Unmute video','mute');

      const volume=document.createElement('input');volume.type='range';volume.min='0';volume.max='1';volume.step='0.05';volume.value='1';volume.className='rh-volume';volume.setAttribute('aria-label','Video volume');

      tools.append(playButton,timeline,time,muteButton,volume);frame.append(video,playOverlay,spinner,fullscreen,tools);

      

      const caption=document.createElement('button');caption.type='button';caption.className='rh-card-caption';caption.setAttribute('aria-label','Play '+(item.title || 'video'));

      const title=document.createElement('span');title.className='rh-card-title';title.textContent=item.title || 'Untitled video';

      const meta=document.createElement('span');meta.className='rh-card-meta';meta.textContent=dateText(item.date)+(item.size ? ' · '+formatBytes(item.size) : '');

      caption.append(title,meta);article.append(frame,caption);


      const toggle=()=>{if(video.paused)attemptPlay(video,playOverlay);else video.pause();};

      playOverlay.addEventListener('click',event=>{event.stopPropagation();toggle();});

      video.addEventListener('click',toggle);

      playButton.addEventListener('click',event=>{event.stopPropagation();toggle();});

      caption.addEventListener('click',()=>{if(video.paused)attemptPlay(video,playOverlay);});

      muteButton.addEventListener('click',event=>{event.stopPropagation();video.muted=!video.muted;});

      volume.addEventListener('input',()=>{video.volume=Number(volume.value);if(video.volume>0)video.muted=false;});

      fullscreen.addEventListener('click',event=>{event.stopPropagation();if(document.fullscreenElement)document.exitFullscreen();else if(frame.requestFullscreen)frame.requestFullscreen();else if(video.webkitEnterFullscreen)video.webkitEnterFullscreen();});

      timeline.addEventListener('input',()=>{if(Number.isFinite(video.duration)&&video.duration>0)video.currentTime=(Number(timeline.value)/1000)*video.duration;});

      video.addEventListener('play',()=>{setPlayOverlay(playOverlay,false);playButton.replaceChildren(createIcon('pause'));playButton.setAttribute('aria-label','Pause video');playButton.title='Pause video';});

      video.addEventListener('pause',()=>{playButton.replaceChildren(createIcon('play'));playButton.setAttribute('aria-label','Play video');playButton.title='Play video';if(activeVideo===video)activeVideo=null;});

      video.addEventListener('waiting',()=>{spinner.hidden=false;});

      video.addEventListener('playing',()=>{spinner.hidden=true;});

      video.addEventListener('canplay',()=>{spinner.hidden=true;});

      video.addEventListener('error',()=>{spinner.hidden=true;setPlayOverlay(playOverlay,true);});

      video.addEventListener('timeupdate',()=>{if(video.duration){timeline.value=String(Math.round((video.currentTime/video.duration)*1000));time.textContent=formatTime(video.currentTime)+' / '+formatTime(video.duration);}});

      video.addEventListener('volumechange',()=>{muteButton.replaceChildren(createIcon(video.muted||video.volume===0?'mute':'volume'));muteButton.setAttribute('aria-label',video.muted?'Unmute video':'Mute video');muteButton.title=video.muted?'Unmute video':'Mute video';volume.value=String(video.volume);});

      

      if(observer)observer.observe(article);

      return article;

    }


    function renderFilter(){

      const query=search.value.trim().toLocaleLowerCase();let shown=0;

      grid.querySelectorAll('.rh-card').forEach(card=>{const title=items.get(card.dataset.id)?.title||'';const match=!query||title.toLocaleLowerCase().includes(query);card.style.display=match?'':'none';if(match)shown++;});

      if(query && shown===0 && items.size>0) countLabel.textContent='No videos match your search';

      else countLabel.textContent=items.size+' video'+(items.size===1?'':'s')+' · new uploads appear live';

    }


    async function refreshGallery(){

      if(!apiBase)throw new Error('Missing Google Apps Script Web App URL.');

      const response=await fetch(apiBase,{cache:'no-store'});

      if(!response.ok)throw new Error('Could not load videos from Google Drive.');

      const body=await response.json();

      const videos=Array.isArray(body.videos)?body.videos:[];

      grid.innerHTML = '';

      items.clear();

      videos.forEach(item=>{

        if(!item||item.id==null||!item.url||items.has(String(item.id)))return;

        items.set(String(item.id),item);grid.prepend(createCard(item));

      });

      galleryError.hidden=true;

      renderFilter();

    }


    observer=new IntersectionObserver(entries=>{

      entries.forEach(entry=>ratios.set(entry.target.dataset.id,entry.isIntersecting?entry.intersectionRatio:0));

      let bestId=null,bestRatio=0;

      ratios.forEach((ratio,id)=>{if(ratio>bestRatio){bestRatio=ratio;bestId=id;}});

      if(bestId&&bestRatio>=0.58){const card=grid.querySelector('.rh-card[data-id="'+CSS.escape(bestId)+'"]');const video=card&&card.querySelector('.rh-video');const overlay=card&&card.querySelector('.rh-play-overlay');if(video&&video.paused)attemptPlay(video,overlay);else if(video&&activeVideo!==video){pauseAll(video);activeVideo=video;}}

      else if(activeVideo){activeVideo.pause();}

    },{threshold:[0,0.2,0.4,0.58,0.75,0.9,1]});


    function validateFormInputs() {

      return userNameInput.value.trim() && userPhoneInput.value.trim() && userAddressInput.value.trim() && selectedFile !== null;

    }


    function checkSubmitState() { submit.disabled = !validateFormInputs(); }

    [userNameInput, userPhoneInput, userAddressInput].forEach(input => input.addEventListener('input', checkSubmitState));


    function selectFile(file){

      if(!file)return;

      const ext=(file.name.split('.').pop()||'').toLowerCase();

      if(!['mp4','webm','mov'].includes(ext)){setStatus('Please select an MP4, WebM, or MOV video.','error');return;}

      if(file.size>maxBytes){setStatus('This video is over 50 MB. Choose a smaller file.','error');return;}

      selectedFile=file;

      selectedLabel.style.display='block';

      selectedLabel.textContent=file.name+' · '+formatBytes(file.size);

      if(localPreviewUrl)URL.revokeObjectURL(localPreviewUrl);

      localPreviewUrl=URL.createObjectURL(file);

      preview.src=localPreviewUrl;

      preview.style.display='block';

      checkSubmitState();

      setStatus('','');

    }


    function clearFile(){

      selectedFile=null;

      fileInput.value='';

      form.reset();

      submit.disabled=true;

      selectedLabel.style.display='none';

      preview.pause();

      preview.removeAttribute('src');

      preview.load();

      preview.style.display='none';

    }


    drop.addEventListener('click',()=>fileInput.click());

    fileInput.addEventListener('change',()=>selectFile(fileInput.files&&fileInput.files[0]));

    ['dragenter','dragover'].forEach(type=>drop.addEventListener(type,e=>{e.preventDefault();drop.classList.add('rh-drag');}));

    ['dragleave','drop'].forEach(type=>drop.addEventListener(type,e=>{e.preventDefault();drop.classList.remove('rh-drag');}));

    drop.addEventListener('drop',e=>selectFile(e.dataTransfer&&e.dataTransfer.files[0]));


    form.addEventListener('submit',event=>{

      event.preventDefault();

      if(!validateFormInputs()) return;


      setStatus('Processing and uploading to Google Drive...', '');

      submit.disabled = true;

      progressWrap.style.display = 'block';

      progressBar.style.width = '50%';

      percent.textContent = 'Uploading...';


      const reader = new FileReader();

      reader.onload = function(e) {

        const base64Bytes = e.target.result.split(',')[1];

        const payload = {

          name: userNameInput.value.trim(),

          phone: userPhoneInput.value.trim(),

          address: userAddressInput.value.trim(),

          filename: selectedFile.name,

          mimeType: selectedFile.type || 'video/mp4',

          bytes: base64Bytes

        };


        fetch(apiBase, {

          method: 'POST',

          body: JSON.stringify(payload)

        })

        .then(res => res.json())

        .then(async data => {

          if(data.status === 'success') {

            setStatus('Video uploaded successfully to Google Drive!', 'success');

            clearFile();

            progressBar.style.width = '100%';

            percent.textContent = '100%';

            showView('gallery');

            await refreshGallery();

          } else {

            setStatus('Upload failed: ' + (data.message || 'Unknown error'), 'error');

            submit.disabled = false;

          }

        })

        .catch(err => {

          setStatus('Network error during upload.', 'error');

          submit.disabled = false;

        });

      };

      reader.readAsDataURL(selectedFile);

    });


    refreshGallery().catch(error => {

      galleryError.textContent = error.message || 'Could not load videos.';

      galleryError.hidden = false;

      countLabel.textContent = '';

    });


    window.setInterval(() => refreshGallery().catch(() => {}), 20000);

  })();
